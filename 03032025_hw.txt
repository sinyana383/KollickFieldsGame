    class Audit
    {
        int roomNB;
        int placeNB;
        bool isMultimedia;
        bool isOccupied;
        string cafedra;

        public Audit(int roomNB, int placeNB = 100, bool isMultimedia = true, bool isOccupied = false, string cafedra = "KB-3")
        {
            this.roomNB = roomNB;
            this.placeNB = placeNB;
            this.isMultimedia = isMultimedia;
            this.isOccupied = isOccupied;
            this.cafedra = cafedra;
        }

        public void SetRoomNB(int newRoomNB)
        { 
            if (newRoomNB <= 0) 
            {
                Console.WriteLine("Неправильный формат номера!\n");
                return;
            }

            this.roomNB = newRoomNB;
        }
        public int GetRoomNB() => this.roomNB;
        public void SetOccupied(bool isOccupied) => this.isOccupied = isOccupied;
        public void OccupyRoom(string cafedra, int placeNB) 
        {
            if (isOccupied) 
            {
                Console.WriteLine("Аудитория занята!\n");
                return;
            }
            if (this.placeNB < placeNB)
            {
                Console.WriteLine("Недостаточно мест!\n");
                return;
            }
            if (cafedra != this.cafedra)
            {
                Console.WriteLine("Не ваша кафедра!\n");
                return;
            }

            this.cafedra = cafedra;
            SetOccupied(true);
        }
        public void SwitchMultimedia() => this.isMultimedia = !this.isMultimedia;
        public string Display() 
        {
            string oc = this.isOccupied ? "occupied" : "free";
            string output = $"" +
                $"{this.roomNB} = roomNB;\r\n" +
                $"{this.placeNB} = placeNB;\r\n" +
                $"{this.isMultimedia} = isMultimedia;\r\n" +
                $"{oc} = isOccupied;\r\n" +
                $"{this.cafedra} = cafedra;\r\n" +
                $"--------------------------------------------\n";
            Console.WriteLine(output);
            return output;
        }
    }
